/* ---------------- setup ---------------- */
function setup() {
  var p = PropertiesService.getScriptProperties();
  var ssId = p.getProperty('SHEET_ID'), ss;
  if (ssId) { ss = SpreadsheetApp.openById(ssId); }
  else { ss = SpreadsheetApp.create('Arshees Client Portal'); p.setProperty('SHEET_ID', ss.getId()); }
  Object.keys(TABS).forEach(function (name) {
    var sh = ss.getSheetByName(name) || ss.insertSheet(name);
    if (sh.getLastRow() === 0) { sh.appendRow(TABS[name]); sh.setFrozenRows(1); }
  });
  var first = ss.getSheetByName('Sheet1'); if (first && ss.getSheets().length > 1) ss.deleteSheet(first);
  var fId = p.getProperty('FOLDER_ID');
  if (!fId) { fId = DriveApp.createFolder('Arshees Client Uploads').getId(); p.setProperty('FOLDER_ID', fId); }
  // touch MailApp so its permission is granted during setup
  MailApp.getRemainingDailyQuota();
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'notifyChanges') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('notifyChanges').timeBased().everyMinutes(10).create();
  try { UrlFetchApp.fetch('https://api.onesignal.com/ping', { muteHttpExceptions: true }); } catch (e) {}
  Logger.log('Sheet: ' + ss.getUrl());
  Logger.log('Sheet ID: ' + ss.getId());
  Logger.log('Uploads folder: https://drive.google.com/drive/folders/' + fId);
}

/* ---------------- web entry points ---------------- */
function doGet(e) {
  var p = PropertiesService.getScriptProperties();
  if (!p.getProperty('SHEET_ID') || !p.getProperty('FOLDER_ID') || !SpreadsheetApp.openById(p.getProperty('SHEET_ID')).getSheetByName('Onboarding')) setup();
  return json_({ ok: true, service: 'arshees-portal', ready: true, time: new Date().toISOString() });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    var req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var a = req.action;
    if (a === 'requestCode') return json_(withLock_(lock, function () { return requestCode_(req); }));
    if (a === 'verifyCode') return json_(withLock_(lock, function () { return verifyCode_(req); }));
    if (a === 'me') return json_(me_(req));
    if (a === 'submit') return json_(withLock_(lock, function () { return submit_(req); }));
    if (a === 'logout') return json_(withLock_(lock, function () { return logout_(req); }));
    return json_({ ok: false, error: 'unknown_action' });
  } catch (err) {
    log_('error', '', String(err && err.stack || err));
    return json_({ ok: false, error: 'server_error' });
  }
}

/* ---------------- actions ---------------- */
function requestCode_(req) {
  var email = normEmail_(req.email);
  if (!email) return { ok: false, error: 'bad_email' };
  var client = findClientByEmail_(email);
  // Same answer whether or not the email exists, so the portal can't be used to test addresses.
  var generic = { ok: true, sent: true };
  if (!client && req.register) {
    var nm = String(req.name || '').trim().slice(0, 80), ph = String(req.phone || '').replace(/[^\d+ ]/g, '').trim().slice(0, 20);
    if (nm.length < 2 || ph.replace(/\D/g, '').length < 7) return { ok: false, error: 'bad_register' };
    if (phoneTaken_(ph, email)) return { ok: false, error: 'phone_taken' };
    var today = new Date().toISOString().slice(0, 10);
    if (rows_(tab_('Onboarding')).filter(function (r) { var d = new Date(r.createdAt); return !isNaN(d) && d.toISOString().slice(0, 10) === today; }).length >= CFG.maxRegistrationsPerDay) return { ok: false, error: 'too_many' };
    var cid = 'reg-' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6), nowIso = new Date().toISOString();
    tab_('Clients').appendRow([cid, email, txt_(nm), true, JSON.stringify({ name: nm, phone: ph, email: email, venture: CFG.brand, details: { name: nm, phone: ph, email: email }, sites: [], docs: [], pay: [], balance: [], payments: [] }), nowIso]);
    tab_('Onboarding').appendRow([cid, email, txt_(nm), txt_(ph), 'registered', '', '', '', '', '', '', '', '', '', '', 'registered', '', nowIso, nowIso]);
    log_('registered', email, cid);
    client = findClientByEmail_(email);
  }
  if (!client) { log_('code_unknown_email', email, ''); return generic; }
  var sh = tab_('Codes'), rows = rows_(sh), now = Date.now();
  var mine = rows.filter(function (r) { return r.email === email; });
  var recent = mine.filter(function (r) { return now - new Date(r.sentAt).getTime() < 3600e3; });
  if (recent.length >= CFG.maxCodesPerHour) return { ok: false, error: 'too_many' };
  var last = mine.length ? mine[mine.length - 1] : null;
  if (last && now - new Date(last.sentAt).getTime() < CFG.resendSeconds * 1000) return { ok: false, error: 'wait', seconds: CFG.resendSeconds };
  var code = String(Math.floor(100000 + Math.random() * 900000));
  sh.appendRow([email, hash_(email + ':' + code), new Date(now + CFG.codeMinutes * 60e3).toISOString(), 0, new Date(now).toISOString()]);
  var name = client.name || '';
  var html = '<div style="font-family:Arial,sans-serif;max-width:480px;margin:auto;color:#22181c">' +
    '<p>Dear ' + esc_(name) + ',</p>' +
    '<p>Here is your login code for your ' + CFG.brand + ' client portal:</p>' +
    '<p style="font-size:32px;letter-spacing:8px;font-weight:bold;color:#6B1E3A;text-align:center;margin:24px 0">' + code + '</p>' +
    '<p>It works for ' + CFG.codeMinutes + ' minutes. If you didn\'t ask for this code, you can ignore this email.</p>' +
    '<p>If you need any support, please contact us at ' + CFG.support + '.</p>' +
    '<p>Warm regards,<br>Dr Mohammed Mirshad<br>' + CFG.brand + '<br>+971 58 657 6586 · godigital@arshees.com</p></div>';
  sendMail_({ to: email, subject: 'Your login code: ' + code, htmlBody: html, name: CFG.brand, replyTo: CFG.replyTo });
  log_('code_sent', email, '');
  cleanup_();
  return generic;
}

function verifyCode_(req) {
  var email = normEmail_(req.email), code = String(req.code || '').replace(/\D/g, '');
  if (!email || code.length !== 6) return { ok: false, error: 'bad_code' };
  var sh = tab_('Codes'), vals = sh.getDataRange().getValues(), now = Date.now();
  var rowIx = -1;
  for (var i = vals.length - 1; i >= 1; i--) { if (vals[i][0] === email) { rowIx = i; break; } }
  if (rowIx < 0) return { ok: false, error: 'bad_code' };
  var row = vals[rowIx];
  if (new Date(row[2]).getTime() < now) return { ok: false, error: 'expired' };
  if (+row[3] >= CFG.maxAttempts) return { ok: false, error: 'too_many' };
  if (row[1] !== hash_(email + ':' + code)) { sh.getRange(rowIx + 1, 4).setValue(+row[3] + 1); return { ok: false, error: 'bad_code' }; }
  sh.getRange(rowIx + 1, 3).setValue(new Date(0).toISOString()); // one use only
  var client = findClientByEmail_(email);
  if (!client) return { ok: false, error: 'no_access' };
  var token = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
  tab_('Sessions').appendRow([hash_(token), client.clientId, email, new Date(now).toISOString(), new Date(now + CFG.sessionDays * 864e5).toISOString(), new Date(now).toISOString()]);
  log_('login', email, client.clientId);
  return { ok: true, token: token, data: clientData_(client), app: appInfo_(client) };
}

function me_(req) {
  var s = session_(req.token);
  if (!s) return { ok: false, error: 'logged_out' };
  return { ok: true, data: clientData_(s.client), app: appInfo_(s.client) };
}

function submit_(req) {
  var s = session_(req.token);
  if (!s) return { ok: false, error: 'logged_out' };
  var type = String(req.type || '');
  if (['details', 'payment', 'support', 'files', 'register', 'accept', 'resume'].indexOf(type) < 0) return { ok: false, error: 'bad_type' };
  var ob = onboardRow_(s.client.clientId);
  if (type === 'accept') return accept_(s, ob, req.payload || {});
  if (type === 'resume') return resume_(s, ob, req.payload || {});
  var payload = req.payload || {};
  var files = (req.files || []).slice(0, 30), links = [], metaLines = [];
  if (files.length) {
    var folder = clientFolder_(s.client);
    files.forEach(function (f, n) {
      var bytes = Utilities.base64Decode(String(f.b64 || ''));
      if (bytes.length > CFG.maxFileMB * 1048576) return;
      var stamp = Utilities.formatDate(new Date(), 'Asia/Dubai', 'yyyyMMdd-HHmm');
      var safe = String(f.name || ('file' + n)).replace(/[^\w.\- ]/g, '_').slice(0, 80);
      var file = folder.createFile(Utilities.newBlob(bytes, f.mime || 'application/octet-stream', type + '-' + stamp + '-' + safe));
      links.push(file.getUrl());
      var m = (req.fileMeta || [])[n] || {}; metaLines.push([m.category || type, String(m.caption || '').replace(/[|\n]/g, ' ').slice(0, 120), file.getUrl()].join(' | '));
    });
  }
  var id = 'sub-' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
  var stageMsg = null;
  if (type === 'register' && ob) { obSet_(ob, { details: JSON.stringify(payload).slice(0, 45000), files: (metaLines.length ? metaLines : links).join('\n'), stage: 'submitted', notifiedStage: 'submitted', name: payload.name || ob.row.name }); stageMsg = 'submitted'; }
  if (type === 'payment' && ob && ob.row.stage === 'accepted') { obSet_(ob, { stage: 'paid', notifiedStage: 'paid', paidAt: new Date().toISOString() }); stageMsg = 'paid'; }
  tab_('Submissions').appendRow([id, new Date().toISOString(), s.client.clientId, s.email, type, JSON.stringify(payload).slice(0, 45000), (metaLines.length ? metaLines : links).join('\n'), 'New', '']);
  var label = { details: 'updated their details', payment: stageMsg === 'paid' ? 'CONFIRMED and sent the payment screenshot (' + (ob.row.plan || '') + ')' : 'sent a payment screenshot', support: 'asked for support', files: 'uploaded files', register: 'submitted a NEW WEBSITE REQUEST' }[type];
  try {
    sendMail_({
      to: type === 'support' ? CFG.support : CFG.notify,
      subject: 'Client portal: ' + (s.client.name || s.email) + ' ' + label,
      htmlBody: '<p><b>' + esc_(s.client.name || '') + '</b> (' + esc_(s.email) + ') ' + label + ' in the client portal.</p><pre style="white-space:pre-wrap;font-family:Arial">' + esc_(JSON.stringify(payload, null, 2)) + '</pre>' + (links.length ? '<p>Files:<br>' + links.map(function (l) { return '<a href="' + l + '">' + l + '</a>'; }).join('<br>') + '</p>' : '') + '<p>Ref ' + id + '</p>',
      name: CFG.brand + ' portal', replyTo: s.email
    });
  } catch (e) { log_('notify_failed', s.email, String(e)); }
  try {
    var conf = stageMsg === 'submitted' ? ['We received your website request', 'Thank you! We have received your details and our team is preparing a sample website for you. We will notify you as soon as it is ready to view.']
      : stageMsg === 'paid' ? ['Thanks for confirming — your online clinic is under construction', 'Thank you for confirming and for your payment' + (payload.amount ? ' of ₹' + esc_(payload.amount) : '') + '. Your online clinic is now under construction. We will keep you updated here and by email.']
      : null;
    if (conf) { sendClientMail_(s.client, s.email, conf[0], [conf[1]]); sendPush_(s.client, conf[0], conf[1]); addNotice_(s.client.clientId, conf[0]); }
    else conf = { details: ['We received your details', 'Thank you for sending your details. We will update your website and let you know when it is done.'],
      payment: ['We received your payment screenshot', 'Thank you! We received your payment screenshot' + (payload.amount ? ' for ₹' + esc_(payload.amount) : '') + '. We will confirm the payment shortly and it will show in your portal.'],
      support: ['We received your support request', 'Thank you for your message. Our team will reply within 24 hours.'],
      files: ['We received your files', 'Thank you, your files were uploaded.'], register: ['We received your details', 'Thank you for sending your details.'] }[type];
    if (!stageMsg) { sendClientMail_(s.client, s.email, conf[0], [conf[1]]); addNotice_(s.client.clientId, conf[0]); }
  } catch (e) { log_('confirm_failed', s.email, String(e)); }
  log_('submit_' + type, s.email, id);
  return { ok: true, id: id, files: links.length, onboarding: onboardPublic_(s.client.clientId) };
}

function logout_(req) {
  var t = req.token; if (!t) return { ok: true };
  var sh = tab_('Sessions'), vals = sh.getDataRange().getValues(), h = hash_(t);
  for (var i = vals.length - 1; i >= 1; i--) if (vals[i][0] === h) sh.getRange(i + 1, 5).setValue(new Date(0).toISOString());
  return { ok: true };
}

/* ---------------- onboarding (new customers) ---------------- */
function onboardRow_(clientId) {
  var sh = tab_('Onboarding'), v = sh.getDataRange().getValues(), h = v[0];
  for (var i = v.length - 1; i >= 1; i--) if (v[i][0] === clientId) { var o = {}; h.forEach(function (k, j) { o[k] = v[i][j]; }); return { sh: sh, ix: i + 1, head: h, row: o }; }
  return null;
}
function obSet_(ob, patch) {
  patch.updatedAt = new Date().toISOString();
  Object.keys(patch).forEach(function (k) { var j = ob.head.indexOf(k); if (j >= 0) { ob.sh.getRange(ob.ix, j + 1).setValue(typeof patch[k] === 'string' ? txt_(patch[k]) : patch[k]); ob.row[k] = patch[k]; } });
}
function doctorCount_(row) { var d = {}; try { d = JSON.parse(row.details || '{}'); } catch (e) {} var n = (d.doctors && d.doctors.length) || +d.doctorCount || 1; return Math.max(1, Math.min(20, n)); }
function offerOf_(row) {
  var o = {}; try { o = JSON.parse(row.offer || '{}'); } catch (e) {}
  var n = doctorCount_(row), extra = CFG.extraDoctor, plans = (o.plans || CFG.plans).map(function (p) { var q = JSON.parse(JSON.stringify(p)); q.base = p.price; q.extraDoctors = n - 1; q.extraFee = (n - 1) * extra; q.price = p.price + q.extraFee; return q; });
  return { sampleUrl: o.sampleUrl || '', quoteNo: o.quoteNo || '', validUntil: o.validUntil || '', note: o.note || '', doctors: n, extraDoctor: extra, plans: plans, proNote: o.proNote || CFG.proNote, advancePct: o.advancePct || CFG.advancePct, upi: o.upi || CFG.upi, upiName: o.upiName || CFG.upiName };
}
function onboardPublic_(clientId) {
  var ob = onboardRow_(clientId); if (!ob) return null; var r = ob.row, d = {};
  if (r.stage === 'offer') { var oo = {}; try { oo = JSON.parse(r.offer || '{}'); } catch (e) {} if (oo.validUntil && String(oo.validUntil) < today_()) { obSet_(ob, { stage: 'expired' }); r = onboardRow_(clientId).row; } }
  try { d = JSON.parse(r.details || '{}'); } catch (e) {}
  var prev = null; try { prev = r.offer ? JSON.parse(r.offer) : null; } catch (e) {}
  return { stage: r.stage, name: r.name, phone: r.phone, details: d, prevOffer: prev && prev.sampleUrl ? { date: prev.sentAt || '', validUntil: prev.validUntil || '', quoteNo: prev.quoteNo || '' } : null, createdAt: r.createdAt, offer: ['offer', 'accepted', 'paid', 'building', 'live'].indexOf(r.stage) >= 0 ? offerOf_(r) : null, plan: r.plan, payMode: r.payMode, amount: r.amount, signedName: r.signedName, acceptedAt: r.acceptedAt, paidAt: r.paidAt };
}
function accept_(s, ob, p) {
  if (!ob || ob.row.stage !== 'offer') return { ok: false, error: 'not_ready' };
  var offer = offerOf_(ob.row), plan = offer.plans.filter(function (x) { return x.id === p.plan; })[0];
  if (!plan) return { ok: false, error: 'bad_plan' };
  var name = String(p.signedName || '').trim().slice(0, 80);
  if (name.length < 2 || !p.agree || !/^data:image\/png;base64,/.test(String(p.signature || ''))) return { ok: false, error: 'bad_sign' };
  var mode = p.payMode === 'full' ? 'full' : 'advance', amount = mode === 'full' ? plan.price : Math.round(plan.price * offer.advancePct / 100);
  var folder = clientFolder_(s.client);
  var sig = folder.createFile(Utilities.newBlob(Utilities.base64Decode(String(p.signature).split(',')[1]), 'image/png', 'signature-' + (offer.quoteNo || 'quote') + '.png'));
  var now = new Date().toISOString();
  obSet_(ob, { stage: 'accepted', notifiedStage: 'accepted', plan: plan.name, payMode: mode, amount: amount, signedName: name, signature: sig.getUrl(), acceptedAt: now });
  var id = 'sub-' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
  tab_('Submissions').appendRow([id, now, s.client.clientId, s.email, 'accept', JSON.stringify({ plan: plan.name, price: plan.price, basePrice: plan.base, doctors: offer.doctors, extraDoctorFee: plan.extraFee, renewal: plan.renewal, payMode: mode, amount: amount, signedName: name, quoteNo: offer.quoteNo, sampleUrl: offer.sampleUrl, userAgent: String(p.ua || '').slice(0, 200) }), sig.getUrl(), 'New', '']);
  var t = 'Thank you for choosing the ' + plan.name + ' plan', b = 'You signed the quotation' + (offer.quoteNo ? ' ' + offer.quoteNo : '') + ' for the ' + plan.name + ' plan' + (offer.doctors > 1 ? ' for ' + offer.doctors + ' doctors' : '') + ' (₹' + plan.price.toLocaleString('en-IN') + ' for the first year, then ₹' + plan.renewal.toLocaleString('en-IN') + ' a year). Next step: pay ₹' + amount.toLocaleString('en-IN') + ' by UPI in your portal and send us the screenshot.';
  try { sendClientMail_(s.client, s.email, t, [b]); sendPush_(s.client, t, b); addNotice_(s.client.clientId, t); } catch (e) { log_('accept_mail_failed', s.email, String(e)); }
  try { sendMail_({ to: CFG.notify, subject: 'Client portal: ' + (s.client.name || s.email) + ' SIGNED the ' + plan.name + ' plan', htmlBody: '<p><b>' + esc_(s.client.name || '') + '</b> (' + esc_(s.email) + ') accepted ' + esc_(plan.name) + ' — ₹' + plan.price + ' first year, ₹' + plan.renewal + ' renewal. Paying ' + mode + ': ₹' + amount + '.</p><p>Signed as: ' + esc_(name) + '<br>Signature: <a href="' + sig.getUrl() + '">view</a></p><p>Ref ' + id + '</p>', name: CFG.brand + ' portal', replyTo: s.email }); } catch (e) {}
  log_('accept', s.email, plan.name + ' ' + mode + ' ' + amount);
  return { ok: true, id: id, onboarding: onboardPublic_(s.client.clientId) };
}
// Expired sample: the client chooses to continue the old request or start a new one. Their data is always kept.
function resume_(s, ob, p) {
  if (!ob || ob.row.stage !== 'expired') return { ok: false, error: 'not_ready' };
  var now = new Date().toISOString(), handled = String(ob.row.handled || '');
  if (p.choice === 'new') {
    var keep = handled.split(',').map(function (x) { return x.trim(); }).filter(function (x) { return x && x.indexOf('build') !== 0 && x.indexOf('resume') !== 0 && x.indexOf('takedown') !== 0 && x.indexOf('restore') !== 0; });
    keep.push('new-request:' + now.slice(0, 10));
    obSet_(ob, { stage: 'registered', notifiedStage: 'registered', handled: keep.join(','), plan: '', payMode: '', amount: '', signedName: '', signature: '', acceptedAt: '' });
  } else {
    obSet_(ob, { stage: 'resume', notifiedStage: 'resume' });
  }
  tab_('Submissions').appendRow(['sub-' + Date.now().toString(36), now, s.client.clientId, s.email, 'resume', JSON.stringify({ choice: p.choice === 'new' ? 'new' : 'continue' }), '', 'New', '']);
  var t = p.choice === 'new' ? 'Start your new request' : 'We are bringing back your sample website', b = p.choice === 'new' ? 'Your previous details are saved. Review them in your portal, make any changes and submit.' : 'Thank you! We are putting your sample website back online. We will notify you as soon as it is ready to view again.';
  try { sendClientMail_(s.client, s.email, t, [b]); sendPush_(s.client, t, b); addNotice_(s.client.clientId, t); } catch (e) {}
  try { sendMail_({ to: CFG.notify, subject: 'Client portal: ' + (s.client.name || s.email) + (p.choice === 'new' ? ' started a NEW request' : ' wants to CONTINUE their expired sample'), htmlBody: '<p>' + esc_(s.client.name || '') + ' (' + esc_(s.email) + ')</p>', name: CFG.brand + ' portal', replyTo: s.email }); } catch (e) {}
  return { ok: true, onboarding: onboardPublic_(s.client.clientId) };
}
function clientFolder_(c) {
  var root = DriveApp.getFolderById(prop_('FOLDER_ID')), name = (c.name || c.clientId) + ' · ' + c.clientId, it = root.getFoldersByName(name);
  return it.hasNext() ? it.next() : root.createFolder(name);
}

// Runs every 10 minutes: compares each client's latest snapshot with the last one we told them about.
function notifyChanges() {
  var lock = LockService.getScriptLock(); if (!lock.tryLock(5000)) return;
  try {
    try { notifyOnboarding_(); } catch (e) { log_('onboarding_notify_failed', '', String(e)); }
    var st = tab_('State'), sv = st.getDataRange().getValues(), idx = {};
    for (var i = 1; i < sv.length; i++) idx[sv[i][0]] = i;
    activeClients_().forEach(function (c) {
      var raw = String(c.data || ''), h = hash_(raw), row = idx[c.clientId];
      if (row != null && sv[row][1] === h) return;
      var neu = {}; try { neu = JSON.parse(raw || '{}'); } catch (e) { return; }
      if (row == null) { st.appendRow([c.clientId, h, raw.slice(0, 48000), '[]', new Date().toISOString()]); sv.push([c.clientId, h, raw, '[]', '']); idx[c.clientId] = sv.length - 1; return; }
      if (!sv[row][2]) { st.getRange(row + 1, 2, 1, 2).setValues([[h, raw.slice(0, 48000)]]); sv[row][1] = h; sv[row][2] = raw; return; }
      var old = {}; try { old = JSON.parse(sv[row][2]); } catch (e) {}
      var msgs = diff_(old, neu);
      st.getRange(row + 1, 2, 1, 2).setValues([[h, raw.slice(0, 48000)]]);
      sv[row][1] = h; sv[row][2] = raw;
      if (!msgs.length) return;
      sendClientMail_(c, normEmail_(c.email), msgs.length === 1 ? msgs[0] : 'Updates on your account', msgs);
      sendPush_(c, msgs.length === 1 ? 'Update from ' + CFG.brand : msgs.length + ' updates from ' + CFG.brand, msgs.join(' · '));
      msgs.forEach(function (m) { addNotice_(c.clientId, m); });
      st.getRange(row + 1, 5).setValue(new Date().toISOString());
      log_('notified', c.email, msgs.join(' | ').slice(0, 400));
    });
  } finally { lock.releaseLock(); }
}
function diff_(o, n) {

  (n.docs || []).forEach(function (d) {

    else if (p.status !== d.status) out.push(label + ' is now ' + String(d.status).toLowerCase());
  });
  var op = (o.payments || []).length; (n.payments || []).slice(0, Math.max(0, (n.payments || []).length - op)).forEach(function (p) { out.push('Payment of ' + inr(p.amount, p.cur) + ' received — thank you!'); });
  var os = {}; (o.sites || []).forEach(function (s) { os[s.name] = s; });
  (n.sites || []).forEach(function (s) {
    var p = os[s.name];
    if (!p) out.push('Your website ' + s.name + ' was added to your portal');
    else { if (p.status !== s.status) out.push('Your website ' + s.name + ' is now ' + String(s.status).toLowerCase());
      if (p.renewal !== s.renewal && s.renewal) out.push('Next renewal for ' + s.name + ': ' + s.renewal);

function session_(token) {
  if (!token) return null;
  var sh = tab_('Sessions'), vals = sh.getDataRange().getValues(), h = hash_(String(token)), now = Date.now();
  for (var i = vals.length - 1; i >= 1; i--) {
    if (vals[i][0] !== h) continue;
    if (new Date(vals[i][4]).getTime() < now) return null;
    var client = findClientById_(vals[i][1]);
    if (!client || normEmail_(client.email) !== vals[i][2]) return null; // access removed or email changed
    if (now - new Date(vals[i][5]).getTime() > 3600e3) sh.getRange(i + 1, 6).setValue(new Date(now).toISOString());
    return { client: client, email: vals[i][2] };
  }
  return null;
}
